Конечные точки

При возникновении оповещения ntopng экспортирует оповещение в настроенные конечные точки. Создав собственную конечную точку, пользователи могут легко экспортировать данные оповещений в предпочитаемую ими службу или инициировать специализированное действие на основе сработавшего оповещения.

Как они работают

Каждая конечная точка оповещения имеет выделенную очередь Redis FIFO. При срабатывании оповещения ntopng ставит в очередь представление предупреждений JSON для каждой из включенных очередей конечных точек. Периодически ntopng вызывает логика конечной точки, которая отвечает за исключение оповещений из очереди и их обработку.

Определение конечных точек

Конечные точки определены в ./alert_endpoints подкаталог скрипта. Давайте проанализируем email_alert_endpoint в качестве примера.

Скрипт конечной точки

Файл электронная почта.lua содержит фактическую логику конечной точки. Модуль имеет следующую структуру:

  • endpoint.EXPORT_FREQUENCY: определяет частоту вызова конечная точка.dequeueRecipientAlerts. Обычно 60 секунд подходит для большинства практических случаев.

  • endpoint.prio: определяет приоритет выполнения конечная точка.dequeueRecipientAlerts по отношению к другим конечным точкам. Конечные точки с более высоким приоритетом будут вызываться первыми (поэтому они имеют привилегии, особенно когда время строгое).

  • endpoint.onLoad(): может использоваться для программного выполнения определенных действий при загрузке скрипта.

  • endpoint.isAvailable(): можно использовать для программного отключения конечной точки (например, отключить конечную точку на какая-то платформа). Должно возвращать true, если конечная точка в данный момент может использоваться (как только пользователь включит ее из настройки конечных точек) или false, если конечную точку не следует использовать и ее настройки следует скрыть.

  • endpoint.dequeueRecipientAlerts(recipient, budget): вызывается периодически (в зависимости от конечная точка.EXPORT_FREQUENCY). Ожидается, что конечная точка исключит оповещения из очереди из предоставленных получатель.export_queue и обрабатывать их до бюджет. Функция должна вернуть {успех=истина} если оповещения могут быть обработаны правильно, в противном случае {success=false, error_message="что-то пошло не так"} какое-то полезное сообщение об ошибке, о котором будет сообщено пользователю.

  • endpoint.runTest(): он вызывается для проверки конфигурации, например. отправка тестового письма, чтобы убедиться, что оно работает. В случае успеха он должен вернуться ноль, в случае неудачи он должен вернуться message_info, message_severity где сообщение_информация есть локализованное сообщение для отображения на GUI и message_severity — это класс CSS, который применяется к окну сообщения.

Определение предпочтений

Сценарий электронная почта.lua определяет параметры конфигурации для конечной точки и получателя в начале файла:

local email = {
  endpoint_params = {
    { param_name = "smtp_server" },
    { param_name = "email_sender"},
    { param_name = "smtp_username", optional = true },
    { param_name = "smtp_password", optional = true },
  },
  endpoint_template = {
    script_key = "email_alert_endpoint",
    template_name = "email_endpoint.template"
  },
  recipient_params = {
    { param_name = "email_recipient" },
    { param_name = "cc", optional = true },
  },
  recipient_template = {
    script_key = "email_alert_endpoint",
    template_name = "email_recipient.template"
  },
}

Пример

Вот фрагмент с комментариями для конечной точки электронной почты.

local email = {
  endpoint_params = {
    { param_name = "smtp_server" },
    { param_name = "email_sender"},
    { param_name = "smtp_username", optional = true },
    { param_name = "smtp_password", optional = true },
  },
  endpoint_template = {
    script_key = "email_alert_endpoint",
    template_name = "email_endpoint.template"
  },
  recipient_params = {
    { param_name = "email_recipient" },
    { param_name = "cc", optional = true },
  },
  recipient_template = {
    script_key = "email_alert_endpoint",
    template_name = "email_recipient.template"
  },
}

-- email.dequeueRecipientAlerts will be invoked every 60 seconds
email.EXPORT_FREQUENCY = 60

-- It is suggested to bulk multiple alerts into a single message when
-- possible
local MAX_ALERTS_PER_EMAIL = 100

-- ##############################################

function email.isAvailable()
  -- ntop.sendMail is not available on some platforms (e.g. Windows),
  -- so on such platforms this endpoint should be disabled.
  return(ntop.sendMail ~= nil)
end

-- ##############################################

-- This is a custom function defined public with the purpose of allowing
-- other code to call it.
function email.sendEmail(subject, message_body)
  ...

  return ntop.sendMail(from, to, message, smtp_server, username, password)
end

-- ##############################################

-- The function in charge of dequeuing alerts. Some code is boilerplate and
-- can be copied to new endpoints.
function my_endpoint.dequeueRecipientAlerts(recipient, budget)
  local processed = 0

  while processed < budget do
    -- Retrieve a bulk of MAX_ALERTS_PER_EMAIL (or less) alerts
    local alerts = ntop.lrangeCache(recipient.export_queue, 0, MAX_ALERTS_PER_EMAIL-1)

    if not alerts then
      break
    end

    -- Aggregate the alerts into a single message body
    local message_body = {}

    for _, json_message in ipairs(alerts) do
      -- From JSON string to Lua table
      local alert = json.decode(json_message)

      -- Get a standard message for the alert
      message_body[#message_body + 1] = alert_utils.formatAlertNotification(alert, {nohtml=true})
    end

    if email.sendEmail(subject, message_body) then
      -- IMPORTANT: remove the processed messages from the queue
      ntop.ltrimCache(recipient.export_queue, MAX_ALERTS_PER_EMAIL, -1)
    else
      -- NOTE: The messages will be kept into the queue. Export will be
      -- retried at the next round
      return {success=false, error_message="Could not contact the SMTP server"}
    end

        processed = processed + 1
  end

      return {success=true}
end

-- ##############################################

return email

Очень важно удалять из очереди обработанные оповещения (см. ntop.ltrimCache выше) в чтобы освободить место для новых оповещений и избежать их повторной обработки.

Формат оповещения

С помощью alert_utils.formatAlertNotification функции нет необходимости знать формат внутренних оповещений, однако это для того, чтобы выполнить определенные действия на основе предупреждения. Оповещения в очереди имеют следующий формат:

  • ifid: идентификатор интерфейса, на котором было создано предупреждение.

  • action: привлекать, выпускать или магазин. Проверьте оповещения API для получения более подробной информации. [4]

  • alert_tstamp: временная метка Unix, когда было активировано предупреждение.

  • alert_tstamp_end: в случае выпущенных оповещений содержит временную метку Unix события выпуска.

  • alert_type: тип оповещения ИДЕНТИФИКАТОР. alert_consts.alertTypeRaw можно использовать для преобразования его в строку.

  • alert_subtype: необязательный подтип оповещения.

  • alert_severity: серьезность оповещения ИДЕНТИФИКАТОР. оповещениеSeverityRaw можно использовать для преобразования его в строку.

  • alert_json: JSON, который содержит информацию, специфичную для alert_type.

  • alert_entity: объект оповещения ИДЕНТИФИКАТОР. alert_consts.alertEntityRaw можно использовать для преобразования его в строку.

  • alert_entity_val: значение объекта оповещения (например, IP-адрес задействованного хоста).

  • alert_granularity: степень детализации оповещения, то есть частота выполнения проверки оповещений.

Вот пример оповещения о перекрестном пороге минутных пакетов для хоста:

{
  "alert_tstamp": 1585579981,
  "alert_entity": 1,
  "alert_entity_val": "140.82.114.26@0",
  "alert_granularity": 60,
  "action": "engage",
  "alert_type": 32,
  "alert_subtype": "packets",
  "ifid": 1,
  "alert_json": "{\"threshold\":1,\"alert_generation\":{\"subdir\":\"host\",\"script_key\":\"packets\",\"confset_id\":0},\"operator\":\"gt\",\"value\":12,\"metric\":\"packets\"}",
  "alert_severity": 2,
  "alert_tstamp_end": 1585579981
}

Эту информацию можно использовать для выполнения настраиваемых действий при возникновении предупреждения. Следующий пример показывает как войти в консоль Оповещения злоумышленников о потоке потока.

local my_endpoint = {
  endpoint_params = {
  },
  endpoint_template = {
    script_key = "my_endpoint_alert_endpoint",
    template_name = "my_endpoint_endpoint.template"
  },
  recipient_params = {
  },
  recipient_template = {
    script_key = "my_endpoint_alert_endpoint",
    template_name = "my_endpoint_recipient.template"
  },
}

my_endpoint.EXPORT_FREQUENCY = 60

function my_endpoint.dequeueRecipientAlerts(recipient, budget)
  local alert_consts = require("alert_consts")
  local alert_utils = require("alert_utils")
  local processed = 0

  while processed < budget do
    -- Process 100 alerts at a time
    local bulk_size = 100
    local alerts = ntop.lrangeCache(recipient.export_queue, 0, bulk_size)

    if not alerts then
      break
    end

    for _, json_message in ipairs(alerts) do
      -- From JSON string to Lua table
      local alert = json.decode(json_message)

      if((alert_consts.alertEntityRaw(alert.alert_entity) == "host") and
        (alert_consts.alertTypeRaw(alert.alert_type) == "alert_flows_flood") and
        (alert.alert_subtype == "flow_flood_attacker")) then
         -- Put your custom action here
         traceError(TRACE_NORMAL, TRACE_CONSOLE, "Flow Flood Attacker: " .. alert_utils.formatAlertNotification(alert, {nohtml=true}))
      end
    end

    -- IMPORTANT: remove the processed messages from the queue
    ntop.ltrimCache(recipient.export_queue, bulk_size, -1)

    processed = processed + 1
  end

      return {success=true}
end

return my_endpoint
