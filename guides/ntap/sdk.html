Разработка приложений на основе nTap

nTap можно встроить в приложения, отличные от ntop, с помощью nTap SDK, что с помощью простого API позволяет создавать собственные приложения-сборщики.

SDK является частью пакета nTap и после установки поставляется с:

  • /usr/lib/libntap.a (в некоторых BSD он находится в /usr/local/lib/libntap.a)

  • /usr/include/libntap.h (в случае BSD он находится в /usr/local/include/libntap.h)

Пример приложения можно найти по адресу /usr/share/ntap. Ниже вы можете найти простое приложение-сборщик nTap (libntap_example.c):

#include <stdio.h>
#include "libntap.h"

/* **************************************** */

void process_pkt(unsigned char *pkt, unsigned int pkt_len) {
  printf("Received %u bytes packet\n", pkt_len);
}

/* **************************************** */

int main(int argc, char *argv[]) {
  void *ntap_info;
  unsigned int listening_port = 1234;
  const char *password = "hello";

  if(ntaplib_init(password, listening_port, &ntap_info) != 0) {
    printf("WARNING: Intialization error\n");
    return(-1);
  } else
    printf("Listening for packets on UDP port %u\n", listening_port);

  while(ntaplib_poll(ntap_info, process_pkt) > 0) {
    /*  printf("Processed packet"); */
  }

  printf("Terminating ntap library\n");

  ntaplib_term(ntap_info);
  return(0);
}
